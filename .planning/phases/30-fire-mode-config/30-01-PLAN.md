class FireMode(Enum):

class Action:

    def __init__(self, ..., velocity_overrides=None, dispatch_interval_overrides=None)

    def get_dispatch_interval(self, gesture_name, direction) -> Optional[float]

    def set_dispatch_interval_overrides(self, overrides)
